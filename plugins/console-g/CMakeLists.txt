# Craft Console G (spec console-g r1): channel strip with filters, 4-band EQ, compressor and expander/gate.
# One plug-in; the host chooses stereo (linked dynamics) or mono through the bus layout.
# Added automatically by the root CMakeLists.txt; JUCE, clap-juce-extensions and Catch2 are fetched there.
project (CraftConsoleG VERSION 0.1.0 LANGUAGES C CXX)

# ---------------------------------------------------------------------------------------------
# The normative tables of the spec are embedded at configure time: every CSV line becomes a string literal in
# generated/console_g/SpecTables.h (no file is read at run time). Editing a CSV re-runs the configure step.
set (CONSOLE_G_TABLE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../spec/console-g/r1/tables)
set (CONSOLE_G_TABLES comp_model_fit comp_static exp_static hf_shelf hmf_bell_target hp lf_shelf lmf_bell lp_target
                      param_tapers)
set (CONSOLE_G_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
set (consoleGHeader "// Generated by plugins/console-g/CMakeLists.txt from spec/console-g/r1/tables/*.csv - do not edit.\n")
string (APPEND consoleGHeader "#pragma once\n\nnamespace craft::consoleg::spec_tables\n{\n")
foreach (tableName IN LISTS CONSOLE_G_TABLES)
    set (csvFile ${CONSOLE_G_TABLE_DIR}/${tableName}.csv)
    set_property (DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${csvFile})
    file (STRINGS ${csvFile} csvLines)
    list (LENGTH csvLines csvCount)
    string (APPEND consoleGHeader "inline constexpr int ${tableName}_count = ${csvCount};\n")
    string (APPEND consoleGHeader "inline const char* const ${tableName}[] = {\n")
    foreach (csvLine IN LISTS csvLines)
        string (REPLACE "\\" "\\\\" csvLine "${csvLine}")
        string (REPLACE "\"" "\\\"" csvLine "${csvLine}")
        string (STRIP "${csvLine}" csvLine)
        string (APPEND consoleGHeader "    \"${csvLine}\",\n")
    endforeach()
    string (APPEND consoleGHeader "};\n\n")
endforeach()
string (APPEND consoleGHeader "} // namespace craft::consoleg::spec_tables\n")
file (WRITE ${CONSOLE_G_GEN_DIR}/console_g/SpecTables.h.tmp "${consoleGHeader}")
configure_file (${CONSOLE_G_GEN_DIR}/console_g/SpecTables.h.tmp ${CONSOLE_G_GEN_DIR}/console_g/SpecTables.h COPYONLY)

# ---------------------------------------------------------------------------------------------
# Host-independent DSP core (header only)
add_library (console_g_dsp INTERFACE)
target_include_directories (console_g_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/source ${CONSOLE_G_GEN_DIR})
target_compile_features (console_g_dsp INTERFACE cxx_std_17)

# ---------------------------------------------------------------------------------------------
if (CRAFT_BUILD_PLUGINS)
    set (CONSOLE_G_SOURCES
        source/plugin/PluginProcessor.cpp
        source/plugin/PluginProcessor.h
        source/plugin/ParameterLayout.cpp
        source/plugin/ParameterLayout.h
        source/ui/PluginEditor.cpp
        source/ui/PluginEditor.h
        source/ui/ConsoleLookAndFeel.cpp
        source/ui/ConsoleLookAndFeel.h
        source/ui/Controls.cpp
        source/ui/Controls.h
        source/ui/Meters.cpp
        source/ui/Meters.h)

    juce_add_plugin (CraftConsoleG

        COMPANY_WEBSITE "https://github.com/felixuhmann/waves-craft"
        PLUGIN_MANUFACTURER_CODE Crft
        PLUGIN_CODE CCnG
        PRODUCT_NAME "Craft Console G"
        BUNDLE_ID "audio.craft.CCnG"
        FORMATS VST3 Standalone
        IS_SYNTH FALSE
        NEEDS_MIDI_INPUT FALSE
        NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS FALSE

        VST3_CATEGORIES Fx EQ Dynamics
        VST3_AUTO_MANIFEST TRUE)

    target_sources (CraftConsoleG PRIVATE ${CONSOLE_G_SOURCES})
    target_include_directories (CraftConsoleG PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/source)

    # Development aid: renders the editor to PNG (not installed / released). Option defined in the root.
    if (CRAFT_BUILD_SNAPSHOT)

        target_sources (console_g_snapshot PRIVATE tools/Snapshot.cpp ${CONSOLE_G_SOURCES})

        target_compile_definitions (console_g_snapshot PRIVATE

            JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_STANDALONE_APPLICATION=1 JUCE_MODAL_LOOPS_PERMITTED=1)
        target_link_libraries (console_g_snapshot PRIVATE console_g_dsp juce::juce_audio_utils juce::juce_gui_basics

    endif()
endif()
